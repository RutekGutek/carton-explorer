cmake_minimum_required(VERSION 3.16)
project(CartonExplorer CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
find_package(Threads REQUIRED)

# --- rdzeń (bez Qt): indeks, zapytania, silnik kopiowania
add_library(carton_core STATIC src/util.cpp src/fastfs.cpp src/index.cpp src/query.cpp src/copyengine.cpp)
target_compile_options(carton_core PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
target_link_libraries(carton_core PUBLIC Threads::Threads)

add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core carton_core)
add_executable(test_copy tests/test_copy.cpp)
target_link_libraries(test_copy carton_core)

# --- aplikacja (Qt6)
find_package(Qt6 QUIET COMPONENTS Widgets DBus)
if(Qt6_FOUND)
  set(CMAKE_AUTOMOC ON)
  file(GLOB UI_SRC src/gfx.cpp src/shell.cpp src/app*.cpp src/main.cpp)
  add_executable(carton ${UI_SRC})
  target_compile_options(carton PRIVATE -Wall -Wno-unused-parameter -Wno-sign-compare)
  target_link_libraries(carton carton_core Qt6::Widgets)
  if(Qt6DBus_FOUND)
    target_link_libraries(carton Qt6::DBus)
    target_compile_definitions(carton PRIVATE CARTON_HAVE_DBUS=1)
  endif()
  install(TARGETS carton RUNTIME DESTINATION bin)
  install(FILES data/carton.desktop DESTINATION share/applications)
  install(FILES data/carton.svg DESTINATION share/icons/hicolor/scalable/apps)
else()
  message(WARNING "Qt6 (Widgets) nie znaleziony - budowany tylko rdzeń i testy")
endif()
